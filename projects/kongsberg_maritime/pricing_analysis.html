import matplotlib.pyplot as plt
import pandas as pd

#pull data
fx_ts = con.execute("""
SELECT
    Year,
    Currency,
    SUM(Line_Amount_NOK) AS revenue_nok
FROM customer_invoices
WHERE Currency IN ('USD', 'EUR', 'NOK')
  AND Year BETWEEN 2016 AND 2025
GROUP BY Year, Currency
ORDER BY Year, Currency
""").fetchdf()

#conversion BNOK to MNOK
fx_ts['revenue_mnok'] = fx_ts['revenue_nok'] / 1_000_000

#create plot
fx_pivot = fx_ts.pivot(index='Year', columns='Currency', values='revenue_mnok').fillna(0)
ax = fx_pivot.plot(kind='bar', stacked=True, figsize=(11, 6))

plt.title('AtlantMar Revenue by Currency (2016-2025)')
plt.xlabel('Year')
plt.ylabel('Revenue (MNOK)')
plt.legend(title='Currency')
plt.tight_layout()
plt.show()
plt.clf()

plot_df = df_pv.copy()

# create indexed series per segment (base year = 100)
plot_df["price_index"] = plot_df.groupby("Item_Group")["avg_unit_price"].transform(
    lambda x: x / x.iloc[0] * 100
)
plot_df["volume_index"] = plot_df.groupby("Item_Group")["volume"].transform(
    lambda x: x / x.iloc[0] * 100
)

for seg in plot_df["Item_Group"].unique():
    temp = plot_df[plot_df["Item_Group"] == seg].sort_values("Year")

    plt.figure(figsize=(8,5))
    plt.plot(temp["Year"], temp["price_index"], marker="o", label="Price index")
    plt.plot(temp["Year"], temp["volume_index"], marker="o", label="Volume index")
    plt.title(f"Price vs Volume Index — Segment {seg} (Base year = 100)")
    plt.xlabel("Year")
    plt.ylabel("Index")
    plt.legend()
    plt.grid(True, alpha=0.3)
    plt.tight_layout()
    plt.show()
